estbeta.mm <- function(y){
  m <- mean(y)
  v <- var(y)
  alpha_mm <- m * ((m * (1 - m)/v)-1)
  beta_mm <- (1 - m)*(((m * (1 - m))/v)-1)
  c(shape1 = alpha_mm, shape2 = beta_mm)
}
