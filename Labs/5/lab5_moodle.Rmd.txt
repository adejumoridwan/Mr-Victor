lambda.0<-3
lambda.hat<-fit$estimate[1]

z<-(lambda.hat-lambda.0)/se.lambda.hat
poss<-pnorm(z)

# Geneva rainfall data

We read the data

```{r eval=FALSE}
rainfall<-read.table("geneva-rainfall.txt",header=TRUE)
```
We define the two periods
```{r eval=FALSE}
per1 <- (rainfall[,"year"] <= 1980)
per2 <- !per1
```



Plot the original time series. Is there visual evidence of a shift?
```{r eval=FALSE}
plot(x=rainfall[per1,"year"],y=rainfall[per1,"Geneva"],xlim=range(rainfall[,"year"]),ylim=range(rainfall[,"Geneva"]),col="red",pch=20,cex=2,
xlab="year",ylab="mm")
points(x=rainfall[per2,"year"],y=rainfall[per2,"Geneva"],col="blue",pch=20,cex=2) 
abline(v=1980)
title("Original Measurements")
```


```{r eval=FALSE}
ylim<-range(rainfall[,"Geneva"])
x<-rainfall[per1,"Geneva"]
y<-rainfall[per2,"Geneva"]
boxplot(x, main="Rainfall",xlim=c(0,3),ylim=ylim,col="red")
boxplot(y,add = TRUE,at = 2,ylim=ylim,col="blue")
```

Welch's two-sided test
```{r eval=FALSE}
tt<-t.test(x=x,y=y,alternative="two.sided")
tt
```

Pooled variance two-sided test
```{r eval=FALSE}
tt.pooled<-t.test(x=x,y=y,alternative="two.sided",var.equal=TRUE)
tt.pooled
```
Wilcox-Mann-Whitney test

```{r eval=FALSE}
tt.w<-wilcox.test(x=x,y=y,alternative="two.sided")
tt.w
```

attach(jura)

Calculate the standard deviation of each group

Anova table

# Goodnes-of-fit tests

## Simulated example

```{r eval=FALSE}
set.seed(19)
y<-rnorm(50)
plot(ecdf(y),main="",verticals=FALSE)
curve(pnorm(x),from=min(y),to=max(y),add=TRUE,lwd =2,col=2)


set.seed(23)
y<-rgamma(100,shape =2, scale=3)
plot(ecdf(y),main="",verticals=FALSE)
curve(pgamma(x,shape =2, scale=3),
   from=min(y),to=max(y),add=TRUE,lwd =2,col="red")
curve(pgamma(x,shape =2, scale=4),
   from=min(y),to=max(y),add=TRUE,lwd =2,col="blue")
```


## Kolmogorov-Smirnov test
```{r eval=FALSE}
ks.test(y, "pgamma", shape=2, scale=3)
ks.test(y, "pgamma", shape=2, scale=4)
```
