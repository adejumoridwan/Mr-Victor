# Prediction and prediction interval

```{r}
CO2.new<-data.frame(CO2=c(370,395))
predict(fit2,CO2.new,interval="confidence",level=0.95)
predict(fit2,CO2.new,interval="prediction",level=0.95)
```


```{r}
plot(CO2,Temp,ylim=c(-30,100))
abline(fit2)
CO2.new<-data.frame(CO2=seq(330,450,by=1))
pred<-predict(fit2,CO2.new, interval="confidence",level=0.95)
segments(x0=CO2.new$CO2,y0=pred[,2],x1=CO2.new$CO2,y1=pred[,3],lwd=2,col="red")
pred<-predict(fit2,CO2.new, interval="prediction",level=0.95)
segments(x0=CO2.new$CO2-0.5,y0=pred[,2],x1=CO2.new$CO2-0.5,y1=pred[,3],lwd=2,col="blue")
```
