# Reading the data

```{r}
nenana<-read.csv("nenana.csv")
plot(Julian.Date~Year,data=nenana,pch=20)
grid()
```

# Simple linear model

```{r}
fit<-lm(Julian.Date~Year,data=nenana)
summary(fit)
plot(Julian.Date~Year,data=nenana,pch=20)
grid()
abline(fit)
```

Residuals

```{r}
plot(fit,which=1,pch=20)
```
```{r}
res<-resid(fit)
attach(nenana)
plot(res~Year,pch=20)
a<-smooth.spline(Year,res)
lines(a,col="red")
grid()
```

# Model with changepoint


```{r}
changepoint<-1970
nenana$cp1970<-pmax(0,nenana$Year-changepoint)
fit.c<-lm(Julian.Date~Year+cp1970,data=nenana)
summary(fit.c)
```

Residuals

```{r}
res<-resid(fit.c)
plot(res~Year,pch=20)
a<-smooth.spline(Year,res)
lines(a,col="red")
grid()
```

```{r}
plot(Julian.Date~Year,data=nenana,pch=20)
abline(fit)
lines(nenana$Year,predict(fit.c),col="red")
grid()
```
