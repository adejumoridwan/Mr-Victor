Exercise 1 — Basic Calculations (x = 3, y = 4)

x <- 3
y <- 4

log(x + y)           # ln(x + y)
x * y                # xy

log10(x * y^2)       # log10(xy²)

2 * x^(1/3) + y^(1/4)  # 2x^(1/3) + y^(1/4)

10^(x - y) + exp(x * y)  # 10^(x-y) + exp(xy)

Exercise 2 — Vector Operations

x <- c(4, 6, 5, 7, 10, 9, 4, 15)

x < 7       # Returns logical vector: TRUE where elements < 7
x == 7      # Returns logical vector: TRUE only where element equals 7
x <- 7      # ASSIGNS 7 to x, overwriting the vector — don't do this if you want to keep x

# Mean two ways
mean(x)
sum(x) / length(x)

# Subvector of elements larger than the mean
sub_x <- x[x > mean(x)]

# Max of subvector
max(sub_x)

Note: x = 7 and x == 7 are very different — the first overwrites x, the second is a comparison.

Exercise 6 — Air Quality Dataset

data(airquality)

# Create a Date column and drop Month and Day
airquality$Date <- as.Date(paste("1973", airquality$Month, airquality$Day, sep = "-"))
airquality$Month <- NULL
airquality$Day   <- NULL

# Rename Temp to Temperature and convert °F to °C
names(airquality)[names(airquality) == "Temp"] <- "Temperature"
airquality$Temperature <- round((airquality$Temperature - 32) * 5/9, 1)

Exercise 7 — Querying the Data Frame

# Hottest and coldest dates
airquality$Date[which.max(airquality$Temperature)]   # Hottest
airquality$Date[which.min(airquality$Temperature)]   # Coldest

# Days with above-average wind speed
sum(airquality$Wind > mean(airquality$Wind, na.rm = TRUE), na.rm = TRUE)

# Rows with no missing values
sum(complete.cases(airquality))

# Days with solar radiation > 300 Langleys
airquality$Date[!is.na(airquality$Solar.R) & airquality$Solar.R > 300]
